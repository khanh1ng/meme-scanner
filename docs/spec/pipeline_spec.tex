\maketitle

\noindent This document is pseudocode only. Every constant carries the measurement
that produced it. Every data source carries its endpoint. A reader with no prior
knowledge of the project can rebuild the pipeline from these blocks in order.

\vspace{4pt}
\noindent Governing rule, enforced by assertions in Module 9: at simulated time
$t$, code may touch only data a live trader would have held at $t$, including
publication delays.

\vspace{4pt}
\noindent Why the guards exist: an earlier build reported Sharpe 1.07 and final
equity \$580,674. It was invalid. The 82 name universe came from asking a language
model to list meme stocks, and a model trained through 2026 already knows the
answer. Rebuilt point-in-time, Sharpe fell to 0.61, below SPY at 0.95.

FUNCTION fetch_short_interest()
    rows = []
    FOR d IN candidate_settlement_dates(2021-01 .. today):

        # probe d-2 .. d+2 to locate the published file.

FUNCTION fetch_social(day d)
    docs = []; after = epoch(d); before = epoch(d) + 86400
    WHILE true:
        r = GET EP_ARCTIC {subreddit: "wallstreetbets", after: after,
                           before: before, limit: 100}
        IF r.status IN {422, 429}: sleep(30); retry up to 5; CONTINUE
        IF empty(r): BREAK
        FOR doc IN r:
            docs.APPEND({doc_id, created_utc, author_hash: sha256(doc.author),
                         score: doc.ups, num_comments, parent_id,
                         body_text: doc.body})     # STORE FULL TEXT
        after = max(created_utc IN r)
    RETURN docs

    # A count cannot separate 100 posts by 100 people from 100 posts by 1
    # account, cannot separate a thesis from a warning, and cannot see whether

    # BLOCKED, do not attempt: reddit.com public JSON returns 403 to automated

\section{Module 2. Quality gates}

Each gate exists because a specific defect was found and measured. Fail loudly.

\begin{lstlisting}
MODULE gates

FUNCTION gate_1_timestamps(left, right)
    # DEFECT FOUND: the price panel carried 05:00:00 while external sources used
    # 00:00:00. The join produced ZERO overlapping days and every social feature
    # silently became zero. The conclusion "social data is useless" was an
    # artifact of the join, not a finding.
    left.index  = normalize_to_calendar_date(tz_naive(left.index))
    right.index = normalize_to_calendar_date(tz_naive(right.index))
    joined = left.join(right)
    overlap = count(rows where right columns are not null)
    PRINT "coverage:", overlap, "of", len(left)
    ASSERT overlap > 0.5 * expected_days   # never proceed on zero coverage
    RETURN joined


FUNCTION gate_2_corporate_actions(row)
    # DEFECT FOUND: GPOR showed a 52,648% 20-day gain, AQB showed 367,817%.
    # These are bankruptcy reorganisations that reused the ticker, leaving a
    # discontinuity in the adjusted series. Removes 11% of detected events.
    IF row.adv20_usd IS null OR row.adv20_usd < 10000:    RETURN false
    IF row.atr_pct IS null OR row.atr_pct <= 0.005
                           OR row.atr_pct >= 1.00:        RETURN false
    IF row.price_raw < 0.10 OR row.price_raw > 500:       RETURN false
    IF abs(row.ret_20d) > 5.00:                           RETURN false
    RETURN true


FUNCTION gate_3_ticker_extraction(text)
    # DEFECT FOUND: upper(whole sentence) turned "will be" into ticker BE and
    # "any" into ANY. 46% of all mentions were ordinary English words.
    cands = regex_findall(text, "\$[A-Z]{1,5}\b | \b[A-Z]{2,5}\b")
    RETURN [c FOR c IN cands IF c IN master AND c NOT IN blocklist]

FUNCTION build_blocklist(all_docs)
    # A real ticker spikes on news. An English word is flat over time.
    FOR tok IN all_tokens:
        burst[tok] = max(daily_count[tok]) / median(daily_count[tok])
    RETURN {tok WHERE burst[tok] < 10 AND total_count[tok] > threshold}
    # separates cleanly at burst = 10
    #
    # A first attempt correlated each token with the daily total and was WRONG:
    # it flagged GME and AMC because those two DRIVE the total. The burst
    # statistic does not have that failure.

FUNCTION decontaminate(mentions_for_ticker, english_word_proxy)
    # residual removal: strip the component explained by overall chatter volume
    beta = regress(mentions_for_ticker ON english_word_proxy).slope
    RETURN mentions_for_ticker - beta * english_word_proxy


FUNCTION gate_4_null_policy(value, policy)
    # DEFECT FOUND: a short-interest gate returned false when data was absent.
    # That silently deleted 2021 through 2024, left 84 trades, and reported a
    # loss of $389. The bug looked exactly like a strategy result.
    IF value IS null:
        PRINT "null count for this gate:", counter
        IF policy == "allow": RETURN true
        IF policy == "block": RETURN false
        FAIL "policy must be declared explicitly"
    RETURN evaluate(value)


FUNCTION gate_5_feature_sign(feature, horizon)
    # DEFECT FOUND: social level entered the composite with a POSITIVE weight
    # while its measured IC is -0.054. The options term used acceleration
    # (IC ~ 0) instead of level (IC +0.061). Fixing it moved expectancy from
    # +1.54% to +2.10%.
    ic = information_coefficient(feature, horizon, universe = NEUTRAL)
    ASSERT sign(assigned_weight) == sign(ic)
    ASSERT abs(ic) > 0.01      # otherwise the feature contributes noise
    RETURN ic


FUNCTION gate_6_solvency(equity)
    # DEFECT FOUND: fixed $10k sizing kept trading after cumulative profit hit
    # -$878,000 on $100,000 of capital. The reported maxDD of -655% is
    # meaningless. Always report dollar profit beside any percentage drawdown.
    IF equity < POS_SIZE: RETURN "HALT_NEW_ENTRIES"
    RETURN "OK"
END MODULE

MODULE features

FUNCTION build_features(symbol s, day t)
    # EVERY line below reads t-1 or earlier. No value from day t appears here.
    f = {}

    # liquidity
    f.adv20_usd   = mean(close_split[t-20:t] * volume[t-20:t])
    f.adv20_share = mean(volume[t-20:t])

    # volatility
    tr            = max(high - low, abs(high - prev_close), abs(low - prev_close))
    f.atr14       = mean(tr[t-14:t])      # used by the trailing stop, Module 8
    f.atr20       = mean(tr[t-20:t])      # used by the volatility screen
    f.atr_pct     = f.atr20 / close_split[t-1]

    # trend and position. All RATIOS, so immune to split adjustment.
    f.ma20        = mean(close_split[t-20:t])
    f.ma50        = mean(close_split[t-50:t])
    f.ath         = cummax(close_split[2016-01 : t-1])
    f.dd_from_ath = close_split[t-1] / f.ath - 1
    f.close       = close_split[t]        # the ONLY value from day t. Used by
                                          # the Module 7 trigger at the close,
                                          # never by the Module 6 universe.
    f.ret_5d      = close_split[t-1] / close_split[t-6]  - 1
    f.ret_20d     = close_split[t-1] / close_split[t-21] - 1
    f.ret_60d     = close_split[t-1] / close_split[t-61] - 1
    f.pct_vs_ma20 = close_split[t-1] / f.ma20 - 1

    # volume behaviour
    f.rvol        = volume[t] / mean(volume[t-20:t])   # evaluated at close t
    f.vol_build   = mean(volume[t-20:t]) / mean(volume[t-60:t])

    # price LEVEL uses the RAW panel, never the adjusted one
    f.price_raw   = close_raw[t-1]

    # short interest, publication lagged
    si            = last row WHERE si.usable_from <= t
    f.dtc         = si.dtc
    f.si_change   = si.short_int / si.prev_si - 1

    # history. An event is only KNOWABLE at its peak, so it becomes usable at
    # peak_date + 1, never at the day the run began.
    f.prior_memes = count(events WHERE event.peak_date + 1 <= t)

    # social and options, Modules 4 and 3b
    soc = aggregate_social(s, t-1)        # Module 4. Fields may be null when
                                          # a source failed; see Module 17.
    f.soc_level        = soc.soc_level
    f.soc_accel        = soc.soc_accel
    f.soc_authors      = soc.soc_authors
    f.concentration    = soc.concentration
    f.st_bull_ratio    = soc.st_bull_ratio
    f.llm_conviction   = soc.llm_conviction
    f.llm_catalyst     = soc.llm_catalyst
    f.llm_coordination = soc.llm_coordination
    f.llm_novelty      = soc.llm_novelty
    f.otm_call_level = mean(otm_call_volume[t-5:t])   # LEVEL not acceleration
    RETURN f


FUNCTION detect_meme_events(symbol s)
    # Used to build prior_memes and to build the label in Module 9.
    events = []; i = 60
    WHILE i < len(close) - 1:
        w = close[i+1 : i+21]                       # forward 20 sessions
        IF max(w) / close[i] - 1 >= 0.50:
            pk = argmax(w)
            events.APPEND({ignition: i, peak: pk, gain: max(w)/close[i] - 1,
                           knowable_from: pk + 1})  # NOT from i
            i = pk + 1                              # no overlapping events
        ELSE: i = i + 1
    RETURN events
END MODULE

\begin{lstlisting}
MODULE llm

# ==========================================================================
# WHAT THE MODEL MAY AND MAY NOT DO
#
# ALLOWED:   read social text that already exists and convert it to numbers.
#            This is a text-processing job.
# FORBIDDEN: generate a candidate list, name tickers from memory, or answer
#            any question about what happened after its training cutoff.
#
# This is the central lesson of the project. The original 82 name universe came
# from asking a model to name potential meme stocks. It already knew the answer.
# That single call invalidated every backtest built on top of it.
# A model may PROCESS the past. It must never RECALL the past.
#
# ARCHITECTURAL RULE, from the v1 design:
#   the deterministic composite is computed FIRST and IN FULL.
#   the LLM is an OVERLAY on the surviving top-N names only.
#   it never sees the other 5,000 symbols and never changes an entry or exit.
#   LLM_ENABLED defaults to FALSE. v1 runs and is evaluated without it.
# ==========================================================================

# ---- STEP 1. FREE PRE-LABELLED SENTIMENT, BEFORE SPENDING ANY TOKEN ------
# StockTwits already ships a Bullish or Bearish label on every message.
# Count the ratio directly. No model call is needed for stance.
FUNCTION free_sentiment(symbol s, day d)
    r = GET EP_STWITS.format(sym=s)
    msgs = [m FOR m IN r.messages IF date(m.created_at) == d]
    bull = count(m WHERE m.entities.sentiment.basic == "Bullish")
    bear = count(m WHERE m.entities.sentiment.basic == "Bearish")
    RETURN {st_count: len(msgs),
            st_bull_ratio: bull / max(bull + bear, 1)}
    # NOTE: this endpoint returns 403 and 429 intermittently. Treat as
    # best-effort. Apply gate_4 with policy "allow" so a null does not
    # silently delete history, which is defect 4.

# ---- STEP 2. LLM ONLY FOR WHAT COUNTING CANNOT PRODUCE -------------------
# Counting gives volume and stance. It cannot give narrative or catalyst.
# Those two are the only reasons to spend a token.
CONST SYS_PROMPT = """
You are a text classifier for financial social media. You will be given recent
posts about one ticker from one day. Return only a JSON object, no prose.
Fields:
  conviction   integer 0 to 3, the strongest level present in the posts.
               0 = passing mentions only
               1 = opinion with no support
               2 = specific reasoning or a price target
               3 = authors state positions held, with size or entry
  catalyst     short phrase naming a specific expected event, for example
               earnings, FDA decision, short squeeze, merger vote, lockup
               expiry, or null if none is named
  coordination true when the posts read as coordinated promotion rather than
               independent opinion: repeated identical slogans, an explicit
               call to buy together, obvious copy paste
  novelty      true when this reads as a new thesis rather than continuation
               of an established one
  time_horizon one of: intraday, days, weeks, months, unclear

You must not use any knowledge of what happened to this stock after these
posts were written. Judge only what the text itself says. Do not name any
ticker that does not appear in the text.
"""

FUNCTION llm_context(symbol s, day d, texts)
    IF NOT LLM_ENABLED: RETURN null

    key = (s, d)
    cache = LOAD LLM_CACHE
    IF key IN cache: RETURN cache[key]      # one call per ticker per day, ever

    payload = {model: LLM_MODEL,
               temperature: LLM_TEMP,
               max_tokens: LLM_MAXTOK,
               messages: [{role: "system", content: SYS_PROMPT},
                          {role: "user",   content: join(texts, limit=8000)}]}
    r = POST EP_LLM payload
        headers {Authorization: "Bearer " + LLM_KEY,
                 Content-Type:  "application/json"}
    TRY:
        out = parse_json(r.choices[0].message.content)
    EXCEPT:
        log_warning("LLM fail", s); RETURN null    # never abort the pipeline

    cache[key] = out; SAVE LLM_CACHE
    RETURN out

# ---- STEP 3. DAILY AGGREGATION ------------------------------------------
FUNCTION aggregate_social(symbol s, day d)
    # d = t-1 strictly. Posts from day t are not reliably available at close t.
    docs = classified documents from calendar day d that mention s

    # counted, free, applies to ALL symbols in the universe
    soc_level     = count(docs)
    soc_authors   = count(distinct author_hash IN docs)
    soc_accel     = soc_level[d] / mean(soc_level[d-20:d]) - 1

FUNCTION build_universe(day t, weights W, thresholds TH)
    scored = []

                                        # eligible_at already applies

        # rank_pct(x, t) ranks x across every tradable symbol on day t only.
        # It looks at no other day, so it is point-in-time safe, and it
        # self-normalises to the volatility regime. See the measurement below
        # for why absolute thresholds were abandoned.
        IF  rank_pct(f.atr_pct, t)     >= TH.ATR_Q      # e.g. top 10 percent
        AND rank_pct(f.ret_20d, t)     >= TH.RET20_Q    # e.g. top 10 percent

        AND f.close                    >  f.ma20:

            # It never looks forward and is always safe.
            # score_with_degradation (Module 17) drops any term whose input
            # is null and renormalises over the survivors. It returns null
            # when too little survives, and that name is simply not ranked.
            terms = [(W.ATR, f.atr_pct),    (W.DD,  -f.dd_from_ath),

                     (W.OPT, f.otm_call_level)]
            score, status = score_with_degradation(terms)
            IF status == "ok": scored.APPEND((score, s))

MODULE layer2        # evaluated at the CLOSE of day t

FUNCTION scan_triggers(universe, day t)
    queue = []
    FOR s IN universe:
        f = build_features(s, t)           # same feature table as Module 6
        typical = (high[t] + low[t] + close[t]) / 3
        IF  f.rvol         >= RVOL_MIN     # unusual participation
        AND f.close        >  typical      # closed in the upper part of range
        AND f.close        >  close[t-1]   # rejects a one day pop
        AND f.close        >  f.ma20       # trend filter
        AND f.ret_5d       >  MOM5_MIN     # rejects sideways drift
        AND f.pct_vs_ma20  <= EXT_CAP      # not already exhausted
        THEN queue.APPEND((f.rvol, s))
    RETURN SORT(queue, descending)

# TIMING, measured: on days that BEGIN a move of +100% or more, median rvol is
# 0.97x, meaning volume is NORMAL. Volume rises AFTER the move starts.
# What DOES precede these moves is vol_build at median 1.28x, a slow month-long
# accumulation. A trigger keyed to a volume spike is therefore structurally
# late, and the accumulation term belongs in Layer 1, not here.


MODULE layer2b       # optional intraday confirmation, next session

FUNCTION confirm_intraday(s)
    # Only symbols that passed layer2 are queried, so this covers ~10 symbols
    # rather than 5,000. That is the whole point of the two layer split.
    FOR bar IN 15min_bars(s, session t+1):
        cum_vol   = sum(volume from open to this bar)
        base      = mean(cum_vol at the SAME time-of-day slot, prior 20 days)
        slot_rvol = cum_vol / base
        # time-of-day adjustment is required. Comparing 10:00 volume to a
        # full-day average flags every stock every morning.
        IF slot_rvol >= 2.0 AND price > vwap: RETURN "ENTER" AT bar.close
        IF bar.time >= 15:30:                 RETURN "STAND_DOWN"
END MODULE

MODULE walkforward

# ---- WHY THE PREVIOUS VALIDATION FAILED ---------------------------------
# 1. An earlier walk-forward split into three folds of EQUAL TRADE COUNT and
#    all three were positive. Splitting by trade count rather than by calendar
#    date concealed that the two most recent years were negative.
# 2. A null test against random ENTRY TIMING passed at p = 0.025 while the
#    universe was invalid, because it held the universe fixed. A control must
#    randomise the same thing the strategy chooses. The strategy chooses NAMES,
#    so the control must choose names randomly.

CONST WARMUP_START = 2016-01-01   # establishes all-time highs only. No trades.
CONST SAMPLE_START = 2021-01-01   # first date where SI and social both exist
CONST TRAIN_MONTHS = 24
CONST TEST_MONTHS  = 6
CONST STEP_MONTHS  = 6
CONST PURGE_MONTHS = 1

# The purge gap is REQUIRED. The label uses a forward 20 session window and
# MAX_HOLD is 60 sessions. Without it a trade opened at the end of training is
# still open during testing and the two windows are not independent.
#
#   fold 1  train 2021-01..2022-12  purge 2023-01  test 2023-02..2023-07
#   fold 2  train 2021-07..2023-06  purge 2023-07  test 2023-08..2024-01
#   fold 3  train 2022-01..2023-12  purge 2024-01  test 2024-02..2024-07
#   fold 4  train 2022-07..2024-06  purge 2024-07  test 2024-08..2025-01
#   fold 5  train 2023-01..2024-12  purge 2025-01  test 2025-02..2025-07
#   fold 6  train 2023-07..2025-06  purge 2025-07  test 2025-08..2026-01
#   fold 7  train 2024-01..2025-12  purge 2026-01  test 2026-02..2026-07

FUNCTION run_walkforward()
    results = []
    FOR fold IN folds:

        # ---------------- IN SAMPLE: fit ----------------
        # Only these may be fitted. Everything in Module 0 marked CONST is
        # fixed for the whole study and is NOT touched here.
        best = null
        FOR W, TH IN parameter_grid(W_ATR, W_DD, W_VB, W_SOC, W_LVL, W_LLM,
                                    W_OPT, N_UNIVERSE, ATR_Q, RET20_Q):
            ASSERT all data touched has date <= fold.train_end
            m = simulate(fold.train_start, fold.train_end, W, TH)
            IF m.sharpe > best.sharpe: best = (W, TH)

        # ---------------- LEAKAGE ASSERTIONS ----------------
        ASSERT no z-score, percentile, or normalisation constant was computed
               across the full sample. All are per-day cross-sectional or
               trailing-window only.
        ASSERT for every test day t, universe(t) used data through t-1 only
        ASSERT every FINRA row used satisfies settle + SI_LAG_DAYS <= t
        ASSERT every social document used has created_utc < t
        ASSERT no position opened during train is still open during test
        ASSERT price LEVEL filters read px_raw, and returns read px_split
        ASSERT symbol list was never modified by hand

        # ---------------- OUT OF SAMPLE: evaluate once ----------------
        strat  = simulate(fold.test_start, fold.test_end, best.W, best.TH)
        floor  = simulate_random(fold.test_start, fold.test_end,
                                 n_per_day = best.TH.N_UNIVERSE)
        ceil   = simulate_biased(fold.test_start, fold.test_end)  # scale only
        bench  = buy_and_hold("SPY", fold.test_start, fold.test_end)
        # all four carry identical costs, identical tradability, identical exits

        results.APPEND({fold, strat, floor, ceil, bench, chosen: best})
    RETURN results

# ---- CONTROLS ------------------------------------------------------------
#   STRATEGY   the pipeline above
#   FLOOR      N_UNIVERSE symbols drawn at random each day from the same
#              tradability-filtered pool. Same trigger, same exits, same costs.
#   CEILING    a deliberately biased universe of names known to have run.
#              For SCALE ONLY. Never reported as an achievable result.
#   BENCHMARK  SPY buy and hold over the identical window.
#
# ---- MEASURED REFERENCE VALUES, same window -----------------------------
#   SPY buy and hold    CAGR +15.9%   Sharpe 0.95   maxDD -25.4%
#   biased 82 names     CAGR +32.7%   Sharpe 1.07   maxDD -19.4%   $580,674
#   screen v1, valid    CAGR +12.9%   Sharpe 0.61   maxDD -36.3%   $212,817
#   random control      CAGR  +3.4%   Sharpe 0.26                  $123,401
# The only admissible version so far sits at 0.61, BELOW SPY at 0.95.
# 0.95 is the number to beat, not zero.

# ---- ACCEPTANCE CRITERIA. Fix these BEFORE running, not after. ----------
FUNCTION accept(results)
    RETURN aggregate_sharpe(results.strat)  > 0.95           # must beat SPY
       AND count(folds with expectancy > 0) >= 5 of 7
       AND aggregate_expectancy(strat) - aggregate_expectancy(floor)
             > 2 * standard_error
       AND aggregate_profit(strat, excluding top 10 trades) > 0
       AND beta_to_spy(strat) < 0.30
       AND median_cost_per_round_trip < gross_expectancy / 3

# Criterion 4 answers the concentration objection directly. The thesis DOES
# depend on large winners and that is legitimate. But a strategy that is
# negative without its ten best trades has no measurable base and cannot be
# separated from luck at this sample size. In the biased run the top 10 trades
# were 71% of all profit.
END MODULE

# ApeWisdom is the only free source that gives a RANK, not just a count.

    rows = []
    FOR page IN 1..5:
        r = GET "https://apewisdom.io/api/v1.0/filter/wallstreetbets/page/" + page
        rows.APPEND(r.results)   # {ticker, mentions, mentions_24h_ago, rank}
    RETURN rows

    # Backtest uses Arctic Shift, live uses this. They are DIFFERENT SERIES.
    # See Module 20, which exists entirely because of this asymmetry.

FUNCTION define_trending(symbol s, day d)
    # three competing definitions. Measure each separately via gate_5.
    # Do NOT blend them before measuring.
    lvl        = mentions[s, d]
    accel      = mentions[s, d] / mean(mentions[s, d-20:d]) - 1
    rank_delta = rank[s, d-1] - rank[s, d]           # positive = climbing

    is_trending_level = lvl >= pct(all_mentions[d], 90)   # cross-sectional
    is_trending_accel = accel >= 2.0                      # 3x the 20 day mean
    is_trending_rank  = rank_delta >= 20 AND rank[s, d] <= 50

    # STATUS: soc_level IC -0.054, soc_accel IC +0.025, rank_delta NEVER
    # measured. None may enter the composite before gate_5 clears it.
    RETURN {is_trending_level, is_trending_accel, is_trending_rank,
            lvl, accel, rank_delta}

    # the REVERSE screen. Drop these first, cheaply, before any scoring.

\section{Module 14. Halts and limit up limit down}

\begin{lstlisting}
MODULE halts

# THIS IS WHY realised stop losses were -11.8% against a nominal -8%, and why
# the worst reached -39.1%. It was the largest omission in the earlier spec.
#
# LULD bands, NMS Plan:
#   Tier 1 (S&P 500, Russell 1000, select ETPs)      band  5%
#   Tier 2 (everything else, so every meme name)     band 10%
#   price in [0.75, 3.00]                            band 20%
#   price below 0.75                    band = min(75%, $0.15 / price)
#   DOUBLED during 09:30-09:45 and 15:35-16:00
#
# A typical meme name at $4 carries a 10% band, 20% at the open. A stop at 8%
# therefore sits INSIDE the band. Price passes through it with no halt at all.

FUNCTION luld_band(price, tier, time_of_day)
    IF tier == 1:              base = 0.05
    ELIF price >= 3.00:        base = 0.10
    ELIF price >= 0.75:        base = 0.20
    ELSE:                      base = min(0.75, 0.15 / price)
    IF time_of_day IN [09:30-09:45] OR [15:35-16:00]: base = base * 2
    RETURN base

FUNCTION handle_halt(position p, halt_event h)
    IF h.type == "LULD_PAUSE":          # minimum 5 minutes, often longer
        mark_position_unfillable(p)     # a stop is NOT protection during a halt
        ON reopen: fill at the AUCTION PRINT, not at the stop level
    IF h.type IN {"T1", "T12"}:         # news pending, SEC suspension
        mark_position_frozen(p)         # T12 frequently precedes delisting
        value at last print, flag for manual review

FUNCTION probable_halt_day(bar, prev_close, tier)
    # Daily bars do not record halts. Infer them.
    RETURN (bar.high - bar.low) / prev_close
           > 2 * luld_band(prev_close, tier, "mid")
    # On a flagged day do NOT assume a clean fill at the stop price. Assume
    # the worse of {open, stop}. Module 8 already does this, which is why its
    # gap branch MUST be evaluated before its low-touch branch.
END MODULE

\section{Module 17. Graceful degradation}

\begin{lstlisting}
MODULE degradation

# Every external source WILL fail. Measured failures in this project:
#   reddit.com public JSON      permanent 403
#   api.stocktwits.com          intermittent 403 and 429
#   iborrowdesk.com             403 until a browser User-Agent was sent
#   yfinance                    429 across all 120 tickers at once
#   Arctic Shift                422 and 429 under load
#   Alpaca free tier            403 on recent SIP data
#
# A pipeline that ABORTS on any one of these never finishes an overnight run.
# A pipeline that SILENTLY SUBSTITUTES ZERO produces a false conclusion, which
# is exactly what happened when the social features all became zero.
# The only safe behaviour is: degrade, record, and REFUSE TO SCORE.

CONST TIER = {
    price_bars:     "REQUIRED",   # no substitute exists, abort the day
    symbol_master:  "REQUIRED",
    short_interest: "OPTIONAL",   # IC ~ 0, weight already near zero
    social:         "OPTIONAL",
    options:        "OPTIONAL",
    llm:            "OPTIONAL",   # LLM_ENABLED defaults false in any case
}

FUNCTION safe_fetch(source, fetch_fn, day d)
    TRY:
        v = fetch_fn(d)
        health[source, d] = "OK"
        RETURN v
    EXCEPT timeout, 4xx, 5xx AS e:
        health[source, d] = "FAIL:" + e
        IF TIER[source] == "REQUIRED":
            HALT "required source unavailable, no trading decision made"
        RETURN null                # null. NOT zero. null is not zero.

FUNCTION score_with_degradation(symbol s, day t)
    terms = []; weight_used = 0
    FOR (feature, weight) IN composite_terms:
        v = features[s, t][feature]
        IF v IS null: CONTINUE          # drop the TERM, never zero it
        terms.APPEND(weight * z(v)); weight_used += abs(weight)
    IF weight_used < 0.60 * total_weight:
        RETURN null, "insufficient_data"   # refuse to rank this name today
    RETURN sum(terms) / weight_used, "ok"  # renormalise over what survived

# Zeroing a missing z-score asserts the name is exactly average on that
# feature. That is a CLAIM, not an absence. Renormalising over the surviving
# terms makes no claim, which is the correct behaviour under missing data.

FUNCTION daily_health_report()
    PRINT per source: rows fetched, coverage vs expected, first and last date
    PRINT symbols scored, symbols refused for insufficient_data
    ASSERT coverage > 0.5 * expected FOR every REQUIRED source
    # gate_1 promoted from a one-off check to a daily operational gate
END MODULE

# THE PROBLEM
# Scanning today's listed symbols and backtesting them over 2021 to 2026 asks
# "how would this strategy have done on the companies that survived". That is
# not the question. A trader in 2021 could buy BBBY, IRNT, SPRT and MULN. All

#   FINRA snapshot 2021-08-31 : 20,123 symbols
#   FINRA snapshot 2026-07-31 : 22,341 symbols
#   present in 2021, absent in 2026 : 8,437 symbols
#   after filtering to plausible common stock tickers : 8,078 symbols

#

#   1. Every FINRA biweekly file is a POINT-IN-TIME SNAPSHOT of every NMS

#      after the ASSETS endpoint has purged them. Verified: BBBY, IRNT, SPRT
#      and MULN all return HTTP 200 with real bars. A batch request mixing
#      live and dead symbols also returns 200.
#
# So the missing piece was never the price data. It was the symbol list.
# ==========================================================================

FUNCTION build_pit_membership()
    # ONE ROW PER SYMBOL PER SNAPSHOT. This is the listing record.
    membership = {}
    FOR f IN all_finra_files(2021-01 .. today):     # 131 snapshots
        symbols = read_column(f, "symbolCode")
        FOR s IN symbols:
            membership[s].APPEND({settle: f.date,
                                  usable_from: f.date + SI_LAG_DAYS})
    RETURN membership

    # was s a listed NMS security at time t, as known at time t?
    snap = last snapshot WHERE snapshot.usable_from <= t
    RETURN s IN membership_at(snap)

    # units and preferred, none of which belong in this universe.
    keep = []

        IF s ENDS WITH "R" AND len(s) == 5:            CONTINUE  # right

           {NYSE, NASDAQ, AMEX, ARCA, BATS}:           CONTINUE
        IF meta IS NOT null AND name_matches_fund_pattern(meta.name): CONTINUE
        keep.APPEND(s)
    RETURN keep
    # A symbol with NO metadata is KEPT, not dropped. Absence from the assets
    # endpoint is exactly the condition this module exists to handle. Dropping
    # it would reintroduce the bias this module removes.

              AND has_bar(s, t)            # actually traded that day
              AND tradable(s, t)}          # Module 5, unchanged

\begin{lstlisting}
MODULE main

# ==========================================================================
# ONE CODE PATH SERVES BOTH BACKTEST AND LIVE.
# If the backtest and the live scanner are two programs, the backtest is not a
# forecast of the live system. Module 20 stage 2 verifies that claim
# empirically. This module makes it true by construction. The only permitted
# difference is where the clock comes from and where the fills come from.
# ==========================================================================

FUNCTION one_trading_day(day t, mode, W, TH, state)
    # mode is "backtest" or "live". Nothing below branches on mode except the
    # two lines explicitly marked.

    # ---- 06:00 ET. Ingest. Every row lands with a publication date. ----
    FOR source IN [price_bars, symbol_master, short_interest, social,
                   options, borrow]:
        safe_fetch(source, fetcher[source], t)          # Module 17
    daily_health_report()                               # Module 17
    # gate_1 runs inside every join the fetchers perform.

    # ---- 08:00 ET. Layer 0 and Layer 1. Data through t-1 ONLY. ----
    universe = build_universe(t, W, TH)                 # Module 6 via 22
    on_universe_rebuild(universe, state.open_positions) # Module 21
    universe_churn_report()                             # Module 21

    # ---- 09:30 to 16:00 ET. Manage what is already open. ----
    manage_positions(t)                                 # Modules 8, 14, 16

    # ---- 09:30 ET. Fill YESTERDAY's queue at TODAY's open. ----
    # The queue was built at yesterday's close. This is the t+1 entry that
    # Module 8 requires, and the reason expectancy is not overstated by 38%.
    IF mode == "live":     submit_entry(s, t) FOR s IN state.queue    # M19
    IF mode == "backtest": place_entries(state.queue, t-1, state.equity)  # M8
